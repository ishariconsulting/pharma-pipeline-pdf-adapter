function txt(v) {
    return v === null || v === undefined ? "" : String(v);
}
function clean(v) {
    return txt(v).replace(/\s+/g, " ").trim();
}
function norm(v) {
    return txt(v)
        .toLowerCase()
        .normalize("NFKD")
        .replace(/[\u0300-\u036f]/g, "")
        .replace(/[®™]/g, "")
        .replace(/\b([a-z][a-z0-9]{3,})tm\b/g, "$1")
        .replace(/[–—‑−]/g, "-")
        .replace(/[^a-z0-9]+/g, " ")
        .replace(/\s+/g, " ")
        .trim();
}
function compact(v) {
    return norm(v).replace(/\s+/g, "");
}
function unique(arr) {
    return [...new Set((arr || []).filter(Boolean))];
}
function field(name) {
    try { return portfolioTable.getField(name); } catch (_) { return null; }
}
function cellText(record, f) {
    if (!f) return "";
    try { return clean(record.getCellValueAsString(f)); } catch (_) { return ""; }
}
function selectText(record, f) {
    if (!f) return "";
    const raw = record.getCellValue(f);
    if (raw && typeof raw === "object" && raw.name) return clean(raw.name);
    return cellText(record, f);
}
function linkNames(record, f) {
    if (!f) return [];
    const raw = record.getCellValue(f);
    if (!Array.isArray(raw)) return [];
    return raw.map(x => clean(x && x.name)).filter(Boolean);
}
function tokenSet(v, stop = new Set()) {
    return new Set(
        norm(v).split(" ").filter(t => t && t.length > 1 && !stop.has(t))
    );
}
function similarity(a, b, stop = new Set()) {
    const A = tokenSet(a, stop);
    const B = tokenSet(b, stop);
    if (!A.size || !B.size) return 0;
    let inter = 0;
    for (const t of A) if (B.has(t)) inter++;
    const union = new Set([...A, ...B]).size;
    const containment = inter / Math.min(A.size, B.size);
    const jaccard = inter / union;
    return 0.7 * containment + 0.3 * jaccard;
}

function normalizedIndication(v) {
    return norm(
        txt(v)
            .replace(/\bfirst[- ]line\b/gi, "1L")
            .replace(/\bsecond[- ]line\b/gi, "2L")
            .replace(/\bthird[- ]line\b/gi, "3L")
            .replace(/\bnon[- ]small cell lung cancer\b/gi, "NSCLC")
            .replace(/\bchronic weight management\b/gi, "obesity weight management")
            .replace(/\bcastration resistant\b/gi, "CRPC")
            .replace(/\bcastration sensitive\b/gi, "CSPC")
    );
}
function indicationSimilarity(a, b) {
    return similarity(normalizedIndication(a), normalizedIndication(b), IND_STOP);
}
function extractCodes(v) {
    const raw = txt(v).toUpperCase().replace(/[–—‑−]/g, "-");
    const patterns = [
        /\bPF-\s?\d{5,9}\b/g,
        /\bMET-\s?\d{2,5}I\b/g,
        /\bVLA\s?\d{2,5}\b/g,
        /\bSAR[- ]?\d{4,9}\b/g,
        /\bSP[- ]?\d{4,9}\b/g
    ];
    let out = [];
    for (const p of patterns) out = out.concat(raw.match(p) || []);
    return unique(out.map(x => x.replace(/\s+/g, "")));
}
function sourceIdentityParts(row) {
    const a = clean(row.asset);
    const parts = [a, row.developmentCode];
    const beforeParen = clean(a.split("(")[0]);
    if (beforeParen) parts.push(beforeParen);
    const paren = /\(([^)]+)\)/g;
    let m;
    while ((m = paren.exec(a)) !== null) parts.push(clean(m[1]));
    for (const p of a.split(/\+|\//g)) parts.push(clean(p.replace(/[()]/g, " ")));
    return unique(parts.map(norm).filter(Boolean));
}
function existingIdentityParts(record, f) {
    const values = [
        cellText(record, f.brand),
        cellText(record, f.molecule),
        cellText(record, f.devCode),
        cellText(record, f.aliases)
    ];
    const parts = [];
    for (const v of values) {
        if (!v) continue;
        parts.push(v);
        for (const p of v.split(/\+|\/|;|\n/g)) parts.push(p);
    }
    return unique(parts.map(norm).filter(Boolean));
}
function identityEvidence(sourceRow, record, f) {
    const sourceParts = sourceIdentityParts(sourceRow);
    const existingParts = existingIdentityParts(record, f);
    const sCodes = new Set(unique([
        ...extractCodes(sourceRow.asset),
        ...extractCodes(sourceRow.developmentCode)
    ]).map(compact));
    const eCodes = new Set(unique([
        ...extractCodes(cellText(record, f.brand)),
        ...extractCodes(cellText(record, f.molecule)),
        ...extractCodes(cellText(record, f.devCode)),
        ...extractCodes(cellText(record, f.aliases))
    ]).map(compact));
    const sharedCodes = [...sCodes].filter(c => eCodes.has(c));
    let exactIdentity = false;
    let exactPartIdentity = false;
    let bestSimilarity = 0;
    for (const s of sourceParts) {
        for (const e of existingParts) {
            if (!s || !e) continue;
            if (s === e) {
                exactPartIdentity = true;
                exactIdentity = true;
            } else if (
                (s.length >= 5 && e.includes(s)) ||
                (e.length >= 5 && s.includes(e))
            ) {
                exactIdentity = true;
            }
            bestSimilarity = Math.max(bestSimilarity, similarity(s, e));
        }
    }
    return {
        sharedCodes,
        codeMatch: sharedCodes.length > 0,
        exactIdentity,
        exactPartIdentity,
        identitySimilarity: bestSimilarity
    };
}
function sourceWholeIdentityCandidates(row) {
    const raw = clean(row.asset);
    const noParen = clean(raw.replace(/\([^)]*\)/g, " "));
    const values = [raw, noParen, row.developmentCode];
    return unique(values.map(norm).filter(Boolean));
}
function existingWholeIdentityCandidates(record, f) {
    const values = [
        cellText(record, f.brand),
        cellText(record, f.molecule),
        cellText(record, f.devCode)
    ];
    const aliases = cellText(record, f.aliases);
    if (aliases) {
        for (const a of aliases.split(/\n|;|,/g)) values.push(clean(a));
    }
    return unique(values.map(norm).filter(Boolean));
}
function wholeIdentityEvidence(sourceRow, record, f) {
    const sourceWhole = sourceWholeIdentityCandidates(sourceRow);
    const existingWhole = existingWholeIdentityCandidates(record, f);
    let exactWholeIdentity = false;
    let wholeIdentitySimilarity = 0;
    for (const a of sourceWhole) {
        for (const b of existingWhole) {
            if (!a || !b) continue;
            if (a === b) exactWholeIdentity = true;
            wholeIdentitySimilarity = Math.max(
                wholeIdentitySimilarity,
                similarity(a, b)
            );
        }
    }
    const sourceCodes = unique([
        ...extractCodes(sourceRow.asset),
        ...extractCodes(sourceRow.developmentCode)
    ]).map(compact);
    const existingCodes = unique([
        ...extractCodes(cellText(record, f.brand)),
        ...extractCodes(cellText(record, f.molecule)),
        ...extractCodes(cellText(record, f.devCode)),
        ...extractCodes(cellText(record, f.aliases))
    ]).map(compact);
    const sharedCodes = sourceCodes.filter(c => existingCodes.includes(c));
    return {
        exactWholeIdentity,
        wholeIdentitySimilarity,
        sourceCodes,
        existingCodes,
        sharedCodes,
        sourceCodeCoverage: sourceCodes.length ? sharedCodes.length / sourceCodes.length : 0,
        existingCodeCoverage: existingCodes.length ? sharedCodes.length / existingCodes.length : 0
    };
}
function hasStrongIdentity(pair) {
    if (!pair) return false;
    if (pair.prog.programmeMatch) return true;
    if (pair.whole.exactWholeIdentity) return true;
    if (
        pair.id.codeMatch &&
        pair.whole.wholeIdentitySimilarity >= 0.60 &&
        pair.whole.sourceCodeCoverage >= 0.50
    ) return true;
    if (
        pair.id.exactIdentity &&
        pair.whole.wholeIdentitySimilarity >= 0.72
    ) return true;
    return false;
}
function hasAnyIdentity(pair) {
    if (!pair) return false;
    return Boolean(
        pair.prog.programmeMatch ||
        pair.whole.exactWholeIdentity ||
        pair.id.codeMatch ||
        (
            pair.id.exactIdentity &&
            pair.whole.wholeIdentitySimilarity >= 0.45
        )
    );
}

function programmeTokensFromSource(indication) {
    const raw = txt(indication).replace(/[–—‑−]/g, "-");
    const tokens = [];
    const paren = /\(([^)]+)\)/g;
    let m;
    while ((m = paren.exec(raw)) !== null) {
        const group = clean(m[1]);
        if (!group) continue;
        if (/^(biologic|orphan|fast track|prime|priority review|rpd)/i.test(group)) continue;
        if (/[A-Za-z]/.test(group) && /\d/.test(group)) {
            tokens.push(compact(group));
        }
    }
    const common = raw.match(
        /\b(?:MM|MEVPRO|FOURLIGHT|HER2CLIMB|TALAPRO|KATSIS|DV|EV|TV|VALOR|MOUNTAINEER|SYMBIOTIC[- A-Z0-9]*|BE6A\s*LUNG|PADL1NK)[- A-Z0-9]*\d+\b/gi
    ) || [];
    for (const x of common) tokens.push(compact(x));
    return unique(tokens.filter(t => t.length >= 3));
}
function programmeTokensFromExisting(record, f) {
    const names = linkNames(record, f.clinicalTrials);
    const out = [];
    for (const name of names) {
        const cleaned = name.replace(/\bNCT\d+\b/gi, " ");
        const pieces = cleaned.split(/[|/;,]/g).map(clean).filter(Boolean);
        for (const p of pieces) out.push(compact(p));
        out.push(compact(cleaned));
    }
    const indicationTokens = programmeTokensFromSource(
        cellText(record, f.indication)
    );
    for (const t of indicationTokens) out.push(t);
    return unique(out.filter(t => t.length >= 3));
}
function programmeMatchEvidence(sourceRow, record, f) {
    const sourceTokens = programmeTokensFromSource(sourceRow.indication);
    const existingTokens = programmeTokensFromExisting(record, f);
    const matches = [];
    for (const s of sourceTokens) {
        for (const e of existingTokens) {
            if (!s || !e) continue;
            if (s === e || e.includes(s) || s.includes(e)) {
                matches.push({source: s, existing: e});
            }
        }
    }
    return {
        sourceTokens,
        existingTokens,
        matches,
        programmeMatch: matches.length > 0
    };
}

function qualifierSet(v) {
    const s = norm(v)
        .replace(/\bfirst line\b/g, "1l")
        .replace(/\bsecond line\b/g, "2l")
        .replace(/\bthird line\b/g, "3l");
    const q = new Set();
    const tests = [
        ["1l", /\b1l\b|\b1\/2l\b/],
        ["2l", /\b2l\b|\b2l\+\b|\b1\/2l\b|\b2l\/3l\b/],
        ["3l", /\b3l\b|\b3l\+\b|\b2l\/3l\b/],
        ["maintenance", /\bmaintenance\b/],
        ["post-transplant", /\bpost transplant\b|\bafter autologous stem cell transplantation\b/],
        ["transplant-ineligible", /\btransplant ineligible\b/],
        ["relapsed-refractory", /\brelapsed\b|\brefractory\b|\brrmm\b/],
        ["adjuvant", /\badjuvant\b/],
        ["neoadjuvant", /\bneoadjuvant\b/],
        ["metastatic", /\bmetastatic\b/],
        ["early", /\bearly\b/],
        ["squamous", /\bsquamous\b/],
        ["non-squamous", /\bnon squamous\b/],
        ["cspc", /\bcspc\b|\bcastration sensitive\b/],
        ["crpc", /\bcrpc\b|\bcastration resistant\b/],
        ["post-cd38", /\bpost cd38\b|\bdaratumumab\b/],
        ["double-class-exposed", /\bdouble class exposed\b/],
    ];
    for (const [label, re] of tests) if (re.test(s)) q.add(label);
    return q;
}
function qualifierEvidence(sourceRow, record, f) {
    const sourceQ = qualifierSet(sourceRow.indication);
    const existingQ = qualifierSet([
        cellText(record, f.indication),
        cellText(record, f.setting),
        cellText(record, f.segment),
        cellText(record, f.subtype)
    ].join(" "));
    const shared = [...sourceQ].filter(x => existingQ.has(x));
    const lineLabels = new Set(["1l","2l","3l"]);
    const sourceLines = [...sourceQ].filter(x => lineLabels.has(x));
    const existingLines = [...existingQ].filter(x => lineLabels.has(x));
    let lineConflict = false;
    if (sourceLines.length && existingLines.length) {
        lineConflict = !sourceLines.some(x => existingLines.includes(x));
    }
    const keySettingLabels = new Set([
        "maintenance","post-transplant","transplant-ineligible","adjuvant",
        "neoadjuvant","cspc","crpc","post-cd38","double-class-exposed",
        "squamous","non-squamous"
    ]);
    const sourceKey = [...sourceQ].filter(x => keySettingLabels.has(x));
    const existingKey = [...existingQ].filter(x => keySettingLabels.has(x));
    const keyConflict =
        sourceKey.length > 0 &&
        existingKey.length > 0 &&
        !sourceKey.some(x => existingKey.includes(x));
    return {
        sourceQualifiers: [...sourceQ],
        existingQualifiers: [...existingQ],
        shared,
        lineConflict,
        keyConflict
    };
}

function stageRank(v){const x=norm(v);if(x.includes("approved"))return 5;if(x.includes("filed")||x.includes("registration"))return 4;if(x.includes("phase 3"))return 3;if(x.includes("phase 2"))return 2;if(x.includes("phase 1"))return 1;return 0;}
function broadStageCategory(v){const x=norm(v);if(!x)return "";if(x.includes("approved")||x.includes("marketed"))return "approved";if(x.includes("filed")||x.includes("registration"))return "registration";if(x.includes("clinical")||x.includes("phase 1")||x.includes("phase 2")||x.includes("phase 3"))return "clinical";if(x.includes("ind cta")||x.includes("preclinical")||x.includes("discovery"))return "preclinical";return "";}
function phaseEvidence(sourceRow,record,f){

    if(phaseComparisonMode==="BROAD_STAGE_ONLY"){

        return {source:sourceStage||source,sourceStage,existing,exact:false,comparable:false,broadExact:Boolean(bs&&be&&bs===be),broadChanged:Boolean(bs&&be&&bs!==be),existingAhead:false,sourceAhead:false};

}

async function sleep(ms) {

}
async function getJson(path) {
    const retryableStatuses = new Set([408, 429, 500, 502, 503, 504]);
    const attempts = 2;
    let lastError = null;
    for (let attempt = 1; attempt <= attempts; attempt++) {
        try {
            const res = await fetch(`${ADAPTER_BASE_URL}${path}`, {
                method: "GET",
                headers: {
                    "Accept": "application/json",
                    "X-Adapter-Key": adapterApiKey,
                },
            });
            const body = await res.text();
            let data;
            try { data = JSON.parse(body); }
            catch (_) {
                throw new Error(
                    `Adapter returned non-JSON. status=${res.status}; ` +
                    `body=${body.slice(0,500)}`
                );
            }
            if (res.ok) return data;
            const err = new Error(
                `Adapter request failed. status=${res.status}; ` +
                `detail=${JSON.stringify(data).slice(0,700)}`
            );
            if (!retryableStatuses.has(res.status) || attempt === attempts) {
                throw err;
            }
            lastError = err;
            await sleep(1500);
        } catch (error) {
            lastError = error;
            if (attempt === attempts) throw error;
            await sleep(1500);
        }
    }
    throw lastError || new Error("Adapter request failed after retry.");
}

function sourceCoarseVariantKey(sourceRow) {
    return [
        norm(sourceRow.asset),

function sourceVariantQualifier(sourceRow) {
    const key = sourceCoarseVariantKey(sourceRow);
    const isCollision = [...sourceVariantCollisionKeys].some(k =>
        compact(k) === compact(key)
    );
    return isCollision ? norm(sourceRow.description || "") : "";
}
function existingVariantQualifier(record) {
    const brand = clean(cellText(record, f.brand));
    const m = brand.match(/\(([^)]+)\)\s*$/);
    return m ? norm(m[1]) : "";
}
function variantEvidence(sourceRow, record) {
    const sourceVariant = sourceVariantQualifier(sourceRow);
    if (!sourceVariant) {
        return {
            required: false,
            exact: false,
            conflict: false,
            sourceVariant: "",
            existingVariant: ""
        };
    }
    const existingVariant = existingVariantQualifier(record);
    const exact =
        sourceVariant &&
        existingVariant &&
        (
            compact(sourceVariant) === compact(existingVariant) ||
            similarity(sourceVariant, existingVariant) >= 0.92
        );
    return {
        required: true,
        exact,
        conflict: !exact,
        sourceVariant,
        existingVariant
    };
}
function scorePair(sourceRow, record) {
    const id = identityEvidence(sourceRow, record, f);
    const whole = wholeIdentityEvidence(sourceRow, record, f);
    const prog = programmeMatchEvidence(sourceRow, record, f);
    const qual = qualifierEvidence(sourceRow, record, f);
    const phase = phaseEvidence(sourceRow, record, f);
    const variant = variantEvidence(sourceRow, record);
    const ind = indicationSimilarity(sourceRow.indication, cellText(record, f.indication));
    let score = 0;
    const reasons = [];
    if (prog.programmeMatch) {
        score += 90;
        reasons.push("linked Clinical Trial / study acronym match");
    }
    if (whole.exactWholeIdentity) {
        score += 55;
        reasons.push("whole asset/programme identity exact");
    }
    if (id.codeMatch) {
        const codeWeight = Math.round(
            35 + 25 * Math.max(
                whole.sourceCodeCoverage,
                whole.existingCodeCoverage
            )
        );
        score += codeWeight;
        reasons.push(`shared development code ${id.sharedCodes.join(", ")}`);
    }
    if (id.exactIdentity && !whole.exactWholeIdentity) {
        score += 24;
        reasons.push("partial asset/molecule identity");
    } else if (!id.exactIdentity) {
        score += Math.round(id.identitySimilarity * 20);
    }
    if (!phase.comparable) {
        if (phase.broadExact) {score += 8; reasons.push("broad stage aligned");}
        else if (phase.broadChanged) reasons.push("broad stage changed; exact phase not inferred");
        else reasons.push("exact phase unavailable from source");
    } else if (phase.exact) {
        score += 18;
        reasons.push("phase exact");
    } else if (phase.existingAhead) {
        score += 10;
        reasons.push("existing regulatory state is ahead of R&D source; no downgrade");
    } else {
        score -= 28;
        reasons.push(`phase differs (${phase.existing || "blank"})`);
    }
    score += Math.round(ind * 34);
    if (ind >= 0.45) reasons.push(`indication similarity ${ind.toFixed(2)}`);
    score += Math.min(qual.shared.length * 8, 24);
    if (qual.shared.length) reasons.push(`shared qualifiers: ${qual.shared.join(", ")}`);
    if (variant.required) {
        if (variant.exact) {
            score += 60;
            reasons.push(`source variant exact: ${variant.sourceVariant}`);
        } else {
            score -= 100;
            reasons.push(
                `source variant conflict: ${variant.sourceVariant} vs ` +
                `${variant.existingVariant || "blank"}`
            );
        }
    }
    if (qual.lineConflict) {
        score -= 38;
        reasons.push("treatment-line conflict");
    }
    if (qual.keyConflict) {
        score -= 28;
        reasons.push("clinical-setting conflict");
    }
    return {
        record,
        score,
        id,
        whole,
        prog,
        qual,
        phase,
        variant,
        indicationSimilarity: ind,
        reasons
    };
}

async function postJson(path, payload) {
    const retryableStatuses = new Set([408, 429, 500, 502, 503, 504]);
    const attempts = 2;
    let lastError = null;
    for (let attempt = 1; attempt <= attempts; attempt++) {
        try {
            const res = await fetch(ADAPTER_BASE_URL + path, {
                method: "POST",
                headers: {
                    "Accept": "application/json",
                    "Content-Type": "application/json",
                    "X-Adapter-Key": adapterApiKey
                },
                body: JSON.stringify(payload)
            });
            const body = await res.text();
            let data;
            try {
                data = JSON.parse(body);
            } catch (_) {
                throw new Error(
                    "Comparator returned non-JSON. status=" + res.status +
                    "; body=" + body.slice(0,500)
                );
            }
            if (res.ok) return data;
            const err = new Error(
                "Comparator request failed. status=" + res.status +
                "; detail=" + JSON.stringify(data).slice(0,700)
            );
            if (!retryableStatuses.has(res.status) || attempt === attempts) {
                throw err;
            }
            lastError = err;
            await sleep(1500);
        } catch (error) {
            lastError = error;
            if (attempt === attempts) throw error;
            await sleep(1500);
        }
    }
    throw lastError || new Error("Comparator request failed after retry.");
}

/* ------------------------------------------------------------
   9d) Controlled Candidate hold-ledger writer
   ------------------------------------------------------------
   Execution remains disabled unless APPLY_CANDIDATE_HOLD_LEDGER_WRITES
   is explicitly changed to true. Even when enabled, this block writes ONLY
   Portfolio Discovery Candidates. It then returns before any queue/source-
   watch router can run. Non-Candidate writes have a separate hard-off gate.
   ------------------------------------------------------------ */
function sortedUniqueIds(values) {
    return [...new Set((values || []).filter(Boolean))].sort();
}
function sameIdSet(a, b) {
    return JSON.stringify(sortedUniqueIds(a)) ===
        JSON.stringify(sortedUniqueIds(b));
}
function descriptorForStableKey(stableKey) {
    return heldDescriptors.find(
        x => x.stableSourceRecordKey === stableKey
    ) || null;
}
function mappingForStableKey(stableKey) {
    return holdLedgerMappings.find(
        x => x.stableSourceRecordKey === stableKey
    ) || null;
}
function executionEvidenceForOperation(operation, executionId) {
    const descriptor = descriptorForStableKey(
        operation.stableSourceRecordKey
    );
    if (!descriptor) {
        throw new Error(
            "CANDIDATE_WRITE_DESCRIPTOR_MISSING stableKey=" +
            operation.stableSourceRecordKey
        );
    }
    return holdEvidenceString(
        operation.stableSourceRecordKey,
        operation.action,
        operation.action === "UPDATE"
            ? "EXACT_ONE"
            : "EXACT_ZERO_CREATE",
        descriptor.holdTypes,
        "CONTROLLED_APPLY",
        executionId
    );
}
async function rerunCreateComparatorForExecution() {
    if (holdLedgerCreatePlan.length === 0) return [];
    const plannedByKey = new Map(
        holdLedgerCreatePlan.map(x => [
            x.stableSourceRecordKey,
            x
        ])
    );
    const descriptors = holdLedgerCreatePlan.map(plan => {
        const descriptor = descriptorForStableKey(
            plan.stableSourceRecordKey
        );
        if (!descriptor) {
            throw new Error(
                "CANDIDATE_WRITE_CREATE_DESCRIPTOR_MISSING stableKey=" +
                plan.stableSourceRecordKey
            );
        }
        return descriptor;
    });
    const rows = descriptors.map(comparatorSourceRowForHold);
    const conflicts = [];
    for (let offset = 0; offset < rows.length; offset += 24) {
        const part = rows.slice(offset, offset + 24);
        const response = await postJson(
            "/compare/portfolio-discovery",
            {
                company: companyName,
                sourceRows: part,
                portfolioRows: comparatorPortfolioRows,
                companyAliases,
                batchRunId:
                    "HOLD_LEDGER_EXECUTION_RECHECK|" +
                    sourceWatchRecordId +
                    "|" +
                    new Date().toISOString()
            }
        );
        if (
            !response ||
            response.readOnly !== true ||
            !Array.isArray(response.candidates) ||
            response.candidates.length !== part.length ||
            Number(response.sourceRowCount) !== part.length ||
            !response.version
        ) {
            conflicts.push({
                reason: "COMPARATOR_EXECUTION_CONTRACT_FAILED",
                offset
            });
            continue;
        }
        for (let i = 0; i < response.candidates.length; i++) {
            const candidate = response.candidates[i] || {};
            const stableKey = clean(candidate.sourceRecordId);
            const plan = plannedByKey.get(stableKey);
            if (!plan) {
                conflicts.push({
                    stableSourceRecordKey: stableKey,
                    reason:
                        "COMPARATOR_EXECUTION_RETURNED_UNKNOWN_SOURCE_KEY"
                });
                continue;
            }
            const suggestedIds = Array.isArray(
                candidate.existingPortfolioRecordIds
            )
                ? candidate.existingPortfolioRecordIds.filter(
                    id => /^rec[A-Za-z0-9]{14}$/.test(id)
                )
                : [];
            if (
                response.version !== plan.comparatorVersion ||
                clean(candidate.discoveryCandidateId) !==
                    clean(
                        plan.proposedFields[
                            "Discovery Candidate ID"
                        ]
                    ) ||
                clean(candidate.classification) !==
                    clean(
                        plan.proposedFields[
                            "Discovery Classification"
                        ]
                    ) ||
                !sameIdSet(
                    suggestedIds,
                    plan.comparatorSuggestedExistingPortfolioRecordIds
                )
            ) {
                conflicts.push({
                    stableSourceRecordKey: stableKey,
                    reason: "COMPARATOR_EXECUTION_DRIFT",
                    plannedComparatorVersion: plan.comparatorVersion,
                    currentComparatorVersion: response.version,
                    plannedDiscoveryCandidateId:
                        plan.proposedFields[
                            "Discovery Candidate ID"
                        ],
                    currentDiscoveryCandidateId:
                        clean(candidate.discoveryCandidateId),
                    plannedClassification:
                        plan.proposedFields[
                            "Discovery Classification"
                        ],
                    currentClassification:
                        clean(candidate.classification),
                    plannedSuggestedExistingPortfolioRecordIds:
                        sortedUniqueIds(
                            plan
                                .comparatorSuggestedExistingPortfolioRecordIds
                        ),
                    currentSuggestedExistingPortfolioRecordIds:
                        sortedUniqueIds(suggestedIds)
                });
            }
        }
    }
    return conflicts;
}
async function executeCandidateHoldLedgerWrites() {
    if (!APPLY_CANDIDATE_HOLD_LEDGER_WRITES) {
        return {
            mode: "DISABLED",
            candidateWritesExecuted: 0
        };
    }
    if (!CANARY_READ_ONLY) {
        throw new Error(
            "CANDIDATE_WRITE_REQUIRES_CANARY_READ_ONLY_TRUE"
        );
    }
    if (ALLOW_NON_CANDIDATE_AIRTABLE_WRITES) {
        throw new Error(
            "CANDIDATE_WRITE_REQUIRES_NON_CANDIDATE_WRITES_DISABLED"
        );
    }
    if (
        !candidateWritePreflightReady ||
        !holdLedgerUpsertPlanReady ||
        holdAssignmentOverlap.length !== 0 ||
        heldExistingDisappearanceOverlap.length !== 0
    ) {
        throw new Error(
            "CANDIDATE_WRITE_PREFLIGHT_NOT_READY"
        );
    }

    // Immediate race-condition check against live Candidate state.
    const freshQuery = await candidateTable.selectRecordsAsync({
        fields: Object.values(hcf)
    });
    const freshByRecordId = new Map(
        freshQuery.records.map(record => [record.id, record])
    );
    const freshByDiscoveryId = new Map();
    const freshSourceByStableKey = new Map();
    for (const record of freshQuery.records) {
        const discoveryId = clean(
            record.getCellValueAsString(hcf.id)
        );
        if (discoveryId) {
            if (!freshByDiscoveryId.has(discoveryId)) {
                freshByDiscoveryId.set(discoveryId, []);
            }
            freshByDiscoveryId.get(discoveryId).push(record);
        }
        const watchIds = linkedIds(record, hcf.sourceWatch);
        if (!watchIds.includes(sourceWatchRecordId)) continue;
        const stableKey = clean(
            record.getCellValueAsString(hcf.sourceRecord)
        );
        if (!stableKey) continue;
        if (!freshSourceByStableKey.has(stableKey)) {
            freshSourceByStableKey.set(stableKey, []);
        }
        freshSourceByStableKey.get(stableKey).push(record);
    }

    const liveConflicts = [];
    for (const plan of holdLedgerUpdatePlan) {
        const record = freshByRecordId.get(
            plan.candidateRecordId
        );
        const mapping = mappingForStableKey(
            plan.stableSourceRecordKey
        );
        if (!record || !mapping) {
            liveConflicts.push({
                action: "UPDATE",
                stableSourceRecordKey:
                    plan.stableSourceRecordKey,
                candidateRecordId: plan.candidateRecordId,
                reason: !record
                    ? "UPDATE_TARGET_MISSING"
                    : "UPDATE_MAPPING_MISSING"
            });
            continue;
        }
        const stableMatches =
            freshSourceByStableKey.get(
                plan.stableSourceRecordKey
            ) || [];
        const discoveryId = clean(
            record.getCellValueAsString(hcf.id)
        );
        const discoveryMatches =
            freshByDiscoveryId.get(discoveryId) || [];
        const currentReviewStatus = clean(
            record.getCellValueAsString(hcf.reviewStatus)
        );
        const currentException = clean(
            record.getCellValueAsString(hcf.exception)
        );
        const currentProgrammeGate = clean(
            record.getCellValueAsString(hcf.programmeGate)
        );
        const currentClassification = clean(
            record.getCellValueAsString(hcf.classification)
        );
        const currentExistingMatchIds =
            linkedIds(record, hcf.existingMatch);
        const retainedExistingMatchIds =
            (plan.retainedFields[
                "Existing Portfolio Match"
            ] || []).map(x => x && x.id).filter(Boolean);
        const identityDrift =
            clean(
                record.getCellValueAsString(hcf.sourceRecord)
            ) !== plan.stableSourceRecordKey ||
            clean(
                record.getCellValueAsString(hcf.asset)
            ) !==
                clean(
                    plan.retainedFields["Source Asset Name"]
                ) ||
            clean(
                record.getCellValueAsString(hcf.indication)
            ) !==
                clean(
                    plan.retainedFields[
                        "Source Indication"
                    ]
                ) ||
            clean(
                record.getCellValueAsString(hcf.phase)
            ) !==
                clean(
                    plan.retainedFields["Source Phase"]
                ) ||
            discoveryId !==
                clean(
                    plan.retainedFields[
                        "Discovery Candidate ID"
                    ]
                ) ||
            currentClassification !==
                clean(
                    plan.retainedFields[
                        "Discovery Classification"
                    ]
                );
        if (
            stableMatches.length !== 1 ||
            stableMatches[0].id !== record.id ||
            discoveryMatches.length !== 1 ||
            linkedIds(record, hcf.company).length !== 1 ||
            linkedIds(record, hcf.company)[0] !==
                targetCompany.id ||
            linkedIds(record, hcf.sourceWatch).length !== 1 ||
            linkedIds(record, hcf.sourceWatch)[0] !==
                sourceWatchRecordId ||
            CLOSED_CANDIDATE_STATUSES.has(
                currentReviewStatus
            ) ||
            currentReviewStatus !==
                mapping.currentReviewStatus ||
            currentException !== mapping.currentException ||
            currentProgrammeGate !==
                mapping.currentProgrammeGate ||
            !sameIdSet(
                currentExistingMatchIds,
                retainedExistingMatchIds
            ) ||
            identityDrift
        ) {
            liveConflicts.push({
                action: "UPDATE",
                stableSourceRecordKey:
                    plan.stableSourceRecordKey,
                candidateRecordId: plan.candidateRecordId,
                reason: "UPDATE_LIVE_STATE_DRIFT",
                currentReviewStatus,
                expectedReviewStatus:
                    mapping.currentReviewStatus,
                currentException,
                expectedException: mapping.currentException,
                currentProgrammeGate,
                expectedProgrammeGate:
                    mapping.currentProgrammeGate,
                currentExistingMatchIds:
                    sortedUniqueIds(
                        currentExistingMatchIds
                    ),
                expectedExistingMatchIds:
                    sortedUniqueIds(
                        retainedExistingMatchIds
                    ),
                identityDrift
            });
        }
    }

    for (const plan of holdLedgerCreatePlan) {
        const stableMatches =
            freshSourceByStableKey.get(
                plan.stableSourceRecordKey
            ) || [];
        const discoveryId = clean(
            plan.proposedFields[
                "Discovery Candidate ID"
            ]
        );
        const discoveryMatches =
            freshByDiscoveryId.get(discoveryId) || [];
        if (
            stableMatches.length !== 0 ||
            discoveryMatches.length !== 0
        ) {
            liveConflicts.push({
                action: "CREATE",

function safeKey(value) {

function parseSourceDateToISO(value) {
    const s = clean(value);
    if (!s) return "";

}

function queueMateriality(delta) {
    const sourcePhase = norm(
        delta.sourcePhase ||
        delta.phase ||
        delta.airtablePhase ||
        ""
    );
    if (delta.classification === "PHASE CHANGE") {
        if (
            sourcePhase.includes("phase 3") ||
            sourcePhase.includes("filed") ||
            sourcePhase.includes("registration") ||
            sourcePhase.includes("approved")
        ) return "High";
        return "Medium";
    }
    if (delta.classification === "SOURCE DISAPPEARANCE") {
        const existingPhase = norm(delta.phase || "");
        if (
            existingPhase.includes("phase 3") ||
            existingPhase.includes("filed") ||
            existingPhase.includes("registration") ||
            existingPhase.includes("approved")
        ) return "High";
        return "Medium";
    }
    if (delta.classification === "NEW PROGRAMME") {
        if (
            sourcePhase.includes("phase 3") ||
            sourcePhase.includes("filed") ||
            sourcePhase.includes("registration") ||
            sourcePhase.includes("approved")
        ) return "High";
        return "Medium";
    }
    return "Medium";
}
function shouldCreateSignal(delta, materiality) {

}
function deltaUpdateId(delta) {
    const prefix = `${companySlug.toUpperCase()}_PIPELINE`;
    if (delta.classification === "PHASE CHANGE") {
        return [
            prefix,
            "PHASE_CHANGE",
            delta.existingRecordId,
            safeKey(delta.previousPhase),
            "TO",
            safeKey(delta.sourcePhase)
        ].join("|");
    }
    if (delta.classification === "INDICATION / PROGRAMME CHANGE") {
        return [
            prefix,
            "PROGRAMME_CHANGE",
            delta.existingRecordId,
            safeKey(delta.sourceIndication)
        ].join("|");
    }
    if (delta.classification === "SOURCE DISAPPEARANCE") {
        return [
            prefix,
            "SOURCE_DISAPPEARANCE",
            delta.recordId
        ].join("|");
    }
    return [
        prefix,
        "NEW_PROGRAMME",
        safeKey(delta.developmentCode || delta.asset),
        safeKey(delta.programmeIndication),
        safeKey(delta.phase),
        safeKey(delta.sourceDescription || "")
    ].filter(Boolean).join("|");
}

        reconciliationStatus === "PASS WITH ROW HOLDS"
            ? allMaterialDeltas.length === 0
                ? `PASS WITH ROW HOLDS - ${companyName} Source Watch refreshed; held rows require review`
                : createdQueueIds.length > 0
                    ? "PASS WITH ROW HOLDS - SAFE DELTAS ROUTED TO INTELLIGENCE UPDATE QUEUE"
                    : "PASS WITH ROW HOLDS - SAFE DELTAS ALREADY QUEUED; held rows require review"
            : allMaterialDeltas.length === 0
                ? `CLEAN RUN - ${companyName} Source Watch refreshed; no queue items required`
                : createdQueueIds.length > 0
                    ? "MATERIAL DELTAS ROUTED TO INTELLIGENCE UPDATE QUEUE"
                    : "MATERIAL DELTAS ALREADY QUEUED - no duplicate queue records created",
    masterDataWrites: 0,
    principle:
        "Pipeline deltas are detected automatically; Portfolio master changes remain controlled through Intelligence Update Queue review."
}, null, 2));


} catch (__sourceError) {
    // SHARED_SOURCE_HOLD_V1: fail closed per source, allow other repeating-group sources to continue.
    const __errorText = String(__sourceError?.message || __sourceError || "Unknown monitoring error")
        .replace(/[\r\n]+/g, " ").slice(0, 600);
    const __sourceId = sourceWatchRecordId;
    const __transient = /\b408\b|\b429\b|\b500\b|\b502\b|\b503\b|\b504\b|timed out|network|fetch failed/i.test(__errorText);
    const __summary = {
        version: VERSION,
        isolationVersion: "SHARED_SOURCE_HOLD_V1",
        sourceWatchRecordId: __sourceId,
        status: __transient ? "DEFERRED - SOURCE TRANSIENT ERROR; OTHER SOURCES CONTINUE"
                            : "FAIL CLOSED - SOURCE RECONCILIATION HOLD; OTHER SOURCES CONTINUE",
        reason: __errorText,
        lastCheckedAdvanced: false,
        masterWrites: 0,
        portfolioWrites: 0
    };
    output.set("genericDeltaSummary", JSON.stringify(__summary, null, 2));
    output.set("genericQueueRouterSummary", JSON.stringify(__summary, null, 2));
    // Operational note is the only failed-source write; preserve prior successful evidence.
    // Do not write Last Checked, Next Check, confidence, Portfolio or Update Queue on error.
    if (
        __sourceId &&
        !CANARY_READ_ONLY &&
        ALLOW_NON_CANDIDATE_AIRTABLE_WRITES
    ) {
        try {
            const __swTable = base.getTable("Source Watch");
            const __record = await __swTable.selectRecordAsync(__sourceId);
            if (__record) {
                const __prior = String(__record.getCellValueAsString("Last Known State / Notes") || "")
                    .replace(/^MONITORING_HOLD[^\n]*\n/, "");
                const __note = "MONITORING_HOLD " + new Date().toISOString()
                    + " | " + (__transient ? "TRANSIENT" : "RECONCILIATION_REVIEW")
                    + " | " + __errorText + "\n" + __prior.slice(0, 3000);
                await __swTable.updateRecordAsync(__sourceId, {
                    "Last Known State / Notes": __note
                });
                __summary.operationalNoteWritten = true;
            }
        } catch (__noteError) {
            __summary.operationalNoteWritten = false;
            __summary.noteError = String(__noteError?.message || __noteError).slice(0, 200);
        }
        output.set("genericQueueRouterSummary", JSON.stringify(__summary, null, 2));
    } else if (CANARY_READ_ONLY) {
        __summary.readOnlyCanary = true;
        __summary.operationalNoteWritten = false;
        output.set("genericQueueRouterSummary", JSON.stringify(__summary, null, 2));
    }
}
