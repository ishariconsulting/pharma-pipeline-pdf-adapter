function clean(v) {
    return v === null || v === undefined
        ? ""
        : String(v).replace(/\s+/g, " ").trim();
}

function norm(v) {
    return clean(v)
        .toLowerCase()
        .normalize("NFKD")
        .replace(/[\u0300-\u036f]/g, "")
        .replace(/[®™]/g, "")
        .replace(/[–—‑−]/g, "-")
        .replace(/[^a-z0-9]+/g, " ")
        .replace(/\s+/g, " ")
        .trim();
}

function splitTerms(v) {
    return clean(v)
        .split(/[\n;|,]+/)
        .map(clean)
        .filter(Boolean);
}

function field(table, name, required = true) {
    let f = null;
    try {
        f = table.getField(name);
    } catch (_) {}

    if (!f && required) {
        throw new Error("Required field missing: " + table.name + "." + name);
    }
    return f;
}

function text(record, f) {
    if (!f) return "";
    try {
        return clean(record.getCellValueAsString(f));
    } catch (_) {
        return "";
    }
}

function linkIds(record, f) {
    if (!f) return [];
    const v = record.getCellValue(f);
    return Array.isArray(v)
        ? v.map(x => x && x.id).filter(Boolean)
        : [];
}

function linkNames(record, f) {
    if (!f) return [];
    const v = record.getCellValue(f);
    return Array.isArray(v)
        ? v.map(x => clean(x && x.name)).filter(Boolean)
        : [];
}

function selectName(record, f) {
    if (!f) return "";
    const v = record.getCellValue(f);
    return v && v.name ? clean(v.name) : text(record, f);
}

async function jsonRequest(url, options = {}, maxAttempts = 2) {
    const retryable = new Set([408, 429, 500, 502, 503, 504]);
    const phase = url.includes("/compare/portfolio-discovery") ? "COMPARATOR" : "EXTRACTION";
    let lastError = null;

    for (let attempt = 1; attempt <= maxAttempts; attempt++) {
        try {
            const res = await fetch(url, options);
            const body = await res.text();
            // Busy and rate-limited services must be deferred until a later
            // controlled Source Watch run rather than immediately retried.
            if (res.status === 408 || res.status === 429 || res.status === 503) {
                const header = res.headers.get("Retry-After");
                const retryAfterSeconds = Math.max(30, Number(header || 30) || 30);
                throw new Error(
                    "DEFERRED_TRANSIENT_HTTP_" + res.status +
                    " phase=" + phase +
                    " retryAfterSeconds=" + retryAfterSeconds +
                    " source retained for controlled retry"
                );
            }

            let data;
            try {
                data = JSON.parse(body);
            } catch (_) {
                const err = new Error(
                    "Non-JSON response. HTTP " + res.status + " phase=" + phase +
                    " body=" + body.slice(0, 500)
                );
                if (attempt < maxAttempts && retryable.has(res.status)) {
                    lastError = err;
                    continue;
                }
                throw err;
            }

            if (res.ok) {
                return data;
            }

            const err = new Error(
                "HTTP " + res.status + " phase=" + phase + " " +
                JSON.stringify(data).slice(0, 900)
            );

            if (attempt < maxAttempts && retryable.has(res.status)) {
                lastError = err;
                continue;
            }

            throw err;
        } catch (e) {
            lastError = e;

            if (attempt >= maxAttempts) {
                break;
            }

            const msg = String(e && e.message || e || "");
            if (msg.includes("DEFERRED_TRANSIENT_HTTP_")) {
                throw e; // Never immediately retry a 429 or 503.
            }
            const looksTransient =
                /408|429|500|502|503|504|timeout|timed out|network/i.test(msg);

            if (!looksTransient) {
                throw e;
            }

            // Airtable automation scripts do not expose setTimeout.
            // Retry immediately; the adapter itself remains fail-closed.
            await Promise.resolve();
        }
    }

    throw lastError || new Error("Request failed after retries.");
}

function stripTrailingAcronym(value) {
    return norm(
        clean(value).replace(
            /\s*\([A-Z0-9-]{2,12}\)\s*$/,
            ""
        )
    );
}

for (let i = 0; i < sourceRows.length; i++) {
    const row = sourceRows[i] || {};
    const sourceRecordId = clean(
        row.sourceRecordId ||
        row.sourceCardOrdinal ||
        row.recordId ||
        String(i + 1)
    );

    const asset = clean(
        row.asset ||
        row.brand ||
        row.molecule ||
        row.developmentCode
    );
    const indication = clean(row.indication);

    const badAsset =
        invalidAssetPrefixes.some(rx => rx.test(asset));

    const badIndication =
        invalidIndicationPrefixes.some(
            rx => rx.test(indication)
        );

    const assetNorm =
        norm(asset);
    const indicationNorm =
        norm(indication);

    const exactAssetIndicationCollapse =
        !!assetNorm &&
        !!indicationNorm &&
        assetNorm.length >= 10 &&
        assetNorm === indicationNorm;

    const acronymOnlyCollapse =
        !!assetNorm &&
        !!indicationNorm &&
        assetNorm.length >= 10 &&
        stripTrailingAcronym(asset) ===
            stripTrailingAcronym(indication) &&
        assetNorm !== indicationNorm;

    const softIndicationIssue =
        semanticReviewIndicationPatterns.some(
            rx => rx.test(indication)
        );

    if (badAsset || badIndication) {
        semanticIdentityIssues.push({
            row: i + 1,
            sourceRecordId,
            asset,
            indication,
            semanticReason:
                badAsset
                    ? "INVALID_ASSET_LABEL"
                    : "INVALID_INDICATION_LABEL",
            phase: clean(
                row.phase || row.sourceStage
            )
        });
        continue;
    }

    if (
        exactAssetIndicationCollapse ||
        acronymOnlyCollapse ||
        softIndicationIssue
    ) {
        const reason =
            exactAssetIndicationCollapse ||
            acronymOnlyCollapse
                ? "ASSET_INDICATION_COLLAPSE_REVIEW"
                : "NON_DISEASE_INDICATION_REVIEW";

        semanticReviewBySourceRecordId.set(
            sourceRecordId,
            {
                reason,
                asset,
                indication,
                phase: clean(
                    row.phase || row.sourceStage
                )
            }
        );
    }
}

/*
Portfolio identity alias bridge V4.4.1
-------------------------------------
Some legacy Portfolio rows represent multiple named components in one
Brand / Asset value, e.g.
"Cenvacibart / REGN7508 and Amrecibart / REGN9933".
Do not rewrite the Portfolio master here. Instead expose only deterministic
"name / development-code" pairs as comparator aliases so current source rows
can resolve to the existing identity without unsafe free-form token splitting.
*/
function portfolioCompositeIdentityAliases(
    asset,
    molecule,
    developmentCode,
    existingAliases
) {
    const out = [];
    const add = value => {
        const v = clean(value);
        if (!v) return;
        if (!out.some(x => norm(x) === norm(v))) {
            out.push(v);
        }
    };

    for (const a of (existingAliases || [])) add(a);
    add(developmentCode);

    for (const label of [asset, molecule]) {
        const raw = clean(label);
        if (!raw) continue;

        const components = raw.split(
            /\s+(?:and|or)\s+|\s*[+±]\s*/i
        );

        for (const component of components) {
            if (!component.includes("/")) continue;

            const pieces = component
                .split(/\s*\/\s*/)
                .map(clean)
                .filter(Boolean);

            const hasDevelopmentCode = pieces.some(piece => {
                const compact = piece.replace(/\s+/g, "");
                return (
                    compact.length >= 4 &&
                    compact.length <= 30 &&
                    /[A-Za-z]/.test(compact) &&
                    /\d/.test(compact) &&
                    !/^NCT/i.test(compact) &&
                    /^[A-Za-z][A-Za-z0-9-]*$/.test(compact)
                );
            });

            if (!hasDevelopmentCode) continue;

            for (const piece of pieces) add(piece);
        }
    }

    return out;
}

/*
Comparator preprocessing V4.4.0
-------------------------------
1) Normalize source asset labels such as "AMRECIBART (REGN9933)" into
   canonical asset + development-code identity before comparison.
2) Split semicolon/newline multi-indication source cards only when every
   segment independently resolves to the controlled indication vocabulary.
   This prevents one source card from collapsing distinct programmes while
   avoiding unsafe splits of free-text qualifiers.
*/

const knownIndicationTerms = [];

    const terms = [meta.name].concat(meta.aliases || []);
    for (const term of terms) {
        const cleaned = clean(term);
        if (!cleaned) continue;
        knownIndicationTerms.push({
            raw: cleaned,
            normalized: norm(cleaned)
        });
    }
}

function isDevelopmentCodeLike(value) {
    const v = clean(value).replace(/\s+/g, "");
    if (!v || v.length < 4 || v.length > 30) return false;
    if (!/[A-Za-z]/.test(v) || !/\d/.test(v)) return false;
    if (/^(phase|stage|dose|age|year|years|mg|mcg|kg|nct)/i.test(v)) {
        return false;
    }
    return /^[A-Za-z][A-Za-z0-9-]*$/.test(v);
}

function normalizeSourceAssetIdentity(row) {
    const rawAsset = clean(
        row.asset ||
        row.brand ||
        row.molecule ||
        row.developmentCode
    );

    if (!rawAsset) {
        return {
            rawAsset: "",
            asset: "",
            developmentCode: clean(row.developmentCode),
            normalized: false,
            extractedCodes: []
        };
    }

    const extractedCodes = [];
    let normalizedAsset = rawAsset;

    normalizedAsset = normalizedAsset.replace(
        /\(([^()]*)\)/g,
        (whole, inner) => {
            const tokens = clean(inner)
                .split(/[,;|/]+/)
                .map(clean)
                .filter(Boolean);

            const codes = tokens
                .map(x => x.replace(/\s+/g, ""))
                .filter(isDevelopmentCodeLike);

            if (!codes.length) {
                return whole;
            }

            for (const code of codes) {
                if (!extractedCodes.includes(code)) {
                    extractedCodes.push(code);
                }
            }

            return " ";
        }
    );

    normalizedAsset = clean(normalizedAsset);

    const existingDevelopmentCode =
        clean(row.developmentCode);

    return {
        rawAsset,
        asset: normalizedAsset || rawAsset,
        developmentCode:
            existingDevelopmentCode ||
            extractedCodes[0] ||
            "",
        normalized:
            norm(normalizedAsset || rawAsset) !== norm(rawAsset) ||
            (!existingDevelopmentCode && extractedCodes.length > 0),
        extractedCodes
    };
}

function indicationSegmentResolves(value) {
    const raw = clean(value);
    if (!raw) return false;

    const withoutTrailingAcronym = clean(
        raw.replace(
            /\s*\(\s*["“”']*[A-Za-z0-9-]{2,12}["“”']*\s*\)\s*$/,
            ""
        )
    );

    const variants = Array.from(
        new Set(
            [norm(raw), norm(withoutTrailingAcronym)]
                .filter(Boolean)
        )
    );

    for (const sourceNorm of variants) {
        const paddedSource = " " + sourceNorm + " ";

        for (const term of knownIndicationTerms) {
            const t = term.normalized;
            if (!t) continue;

            if (sourceNorm === t) {
                return true;
            }

            // Long disease names and short clinical abbreviations such as
            // VTE/ATTR are allowed when they appear as complete tokens.
            if (
                (t.length >= 4 || /^[a-z0-9]{2,6}$/.test(t)) &&
                paddedSource.includes(" " + t + " ")
            ) {
                return true;
            }
        }
    }

    return false;
}

function splitSourceIndication(value) {
    const original = clean(value);
    if (!original) return [""];

    const parts = original
        .split(/\s*;\s*|\n+/)
        .map(clean)
        .filter(Boolean);

    if (parts.length <= 1) {
        return [original];
    }

    if (parts.every(indicationSegmentResolves)) {
        return parts;
    }

    return [original];
}

// Conservative source-scope classifier. This does NOT approve programme matching.
// A single independently supplied NCT identifies study scope; source cards without
// verifiable singularity are never labelled programme-specific just from phase/asset.
function inferSourceScope(c, sourceContext) {
    const ctx = sourceContext || {};
    const indication = clean(c.indication);
    const trialIds = Array.from(new Set(
        (Array.isArray(ctx.trialIds) ? ctx.trialIds : [])
            .map(x => clean(x).toUpperCase())
            .filter(x => /^NCT[0-9]{8}$/.test(x))
    ));
    const full = [indication, clean(ctx.description), clean(ctx.additionalInformation), clean(ctx.study)].join(" ");
    if (/multiple\s+(lines|settings|indications|programmes|programs|cohorts)|various\s+(lines|settings|indications)|across\s+multiple\s+(indications|settings)/i.test(full)) {
        return {scope: "Asset / phase umbrella", why: "Explicit multiple-programme/setting source wording"};
    }
    if (trialIds.length > 1) {
        return {scope: "Asset / phase umbrella", why: `One source card references ${trialIds.length} trials; no unique programme inference`};
    }
    if (trialIds.length === 1 && indication) {
        return {scope: "Specific trial / protocol", why: `Source adapter supplied exactly one valid official NCT: ${trialIds[0]}`};
    }
    return {scope: "Uncertain", why: indication ? "No singular official trial identifier or validated programme-granularity evidence" : "Indication/setting absent from source card"};
}

function structuredProgrammeHoldState(reason, scopeInference) {
    if (!reason) return null;

    if (reason === "SEMANTIC_IDENTITY_QA_REQUIRED") {
        return {
            exception: "Insufficient Identity Evidence",
            next:
                "Re-validate source asset and disease identity from the source structure. A therapy area, mechanism, disease label used as an asset, or other semantic parser artefact cannot become programme-ready.",
            gate:
                "Shared semantic QA → Portfolio Reconciliation Preflight V3.1"
        };
    }

    if (reason === "CO_INFECTION_INDSET_RECONCILIATION_REQUIRED") {
        return {
            exception: "Other",
            next:
                "Resolve every simultaneous co-infection disease component to an active controlled indication, then preview one deterministic PFK1 indset. Do not invent a composite controlled disease.",
            gate:
                "Portfolio Reconciliation Preflight V3.1 → PFK1 Batch Preflight V1.8"
        };
    }

    if (reason === "MULTI_INDICATION_CONTEXT_SPLIT_REQUIRED") {
        return {
            exception: "Multi-Programme / Basket Ambiguity",
            next:
                "Split independent disease contexts into separate programme identities. Do not collapse independent indications into one PFK1 indset.",
            gate:
                "Portfolio Reconciliation Preflight V3.1 → programme-context split → PFK1 Batch Preflight V1.8"
        };
    }

    if (reason === "UMBRELLA_SCOPE_RECONCILIATION_REQUIRED") {
        return {
            exception:
                scopeInference && scopeInference.scope === "Asset / phase umbrella"
                    ? "Multi-Programme / Basket Ambiguity"
                    : "Broad Indication / Source Scope",
            next:
                "Retain the source row as umbrella evidence and reconcile child trial/programme contexts separately before any controlled master review.",
            gate:
                "Shared Portfolio Reconciliation Preflight V3.1"
        };
    }

    return {
        exception: "Other",
        next: "Resolve the shared programme semantic hold before controlled master review.",
        gate: "Shared Portfolio Reconciliation Preflight V3.1"
    };
}

    }

    const sourceContext =
        sourceContextByRecordId.get(
            clean(c.sourceRecordId)
        );

    if (sourceContext) {
        evidenceBits.push(
            "Source context: " +
            JSON.stringify(sourceContext)
        );
    }

    const scopeInference = inferSourceScope(c, sourceContext);

    /*
    Shared programme-semantic safety hold. Staging V4 does not approve
    programme readiness, but it must also clear any stale machine-derived
    Ready gate when semantic identity is unresolved.
    */
    const sourceIndicationForProgramme = clean(c.indication);
    let programmeSemanticHoldReason = "";

    if (semanticReview) {
        programmeSemanticHoldReason =
            "SEMANTIC_IDENTITY_QA_REQUIRED";
    } else if (
        /\bco[- ]?infection\b|\bcoinfection\b/i.test(
            sourceIndicationForProgramme
        )
    ) {
        programmeSemanticHoldReason =
            "CO_INFECTION_INDSET_RECONCILIATION_REQUIRED";
    } else if (
        clean(c.classification) ===
        "MULTI-INDICATION REVIEW"
    ) {
        programmeSemanticHoldReason =
            "MULTI_INDICATION_CONTEXT_SPLIT_REQUIRED";
    } else if (
        scopeInference.scope ===
        "Asset / phase umbrella"
    ) {
        programmeSemanticHoldReason =
            "UMBRELLA_SCOPE_RECONCILIATION_REQUIRED";
    }

    if (programmeSemanticHoldReason) {
        programmeSemanticHoldCount += 1;
        programmeSemanticHoldReasons[
            programmeSemanticHoldReason
        ] =
            (
                programmeSemanticHoldReasons[
                    programmeSemanticHoldReason
                ] || 0
            ) + 1;
        reviewStatus = "Needs Review";
        evidenceBits.push(
            "Programme semantic hold: " +
            programmeSemanticHoldReason +
            ". Resolve through the shared Portfolio Reconciliation Preflight before programme readiness."
        );
    }

    evidenceBits.push("Source scope inference V2: " + scopeInference.scope + " | " + scopeInference.why + ". This is not programme reconciliation approval.");

    evidenceBits.push(
        "Comparator: " +
        clean(comparison.version)
    );

    evidenceBits.push(
        "Adapter: " +
        adapterProfile
    );

    const fields = {};

    const persistedCandidateId =
        existing
            ? text(existing, cf.id)
            : candidateId;
    if (!persistedCandidateId) {
        throw new Error(
            "CANDIDATE_ID_CONTRACT_FAILED blank persisted Candidate ID stableKey=" +
            durableKey
        );
    }

    fields[cf.id.id] =
        persistedCandidateId;

    fields[cf.company.id] =
        [{id: companyId}];

    fields[cf.sourceWatch.id] =
        [{id: sourceWatchRecordId}];

    fields[cf.family.id] =
        {name: "Company Pipeline"};

    fields[cf.classification.id] =
        {name: clean(c.classification)};

    fields[cf.reviewStatus.id] =
        {name: reviewStatus};

    // Preserve any human-reviewed or previously verified scope on upsert.
    const existingScope = existing ? selectName(existing, cf.sourceScope) : "";
    if (!existingScope) {
        fields[cf.sourceScope.id] = {name: scopeInference.scope};
    }

    const structuredHoldState =
        structuredProgrammeHoldState(
            programmeSemanticHoldReason,
            scopeInference
        );

    if (structuredHoldState) {
        const priorException =
            existing && cf.reconciliationException
                ? selectName(existing, cf.reconciliationException)
                : "";
        const priorNext =
            existing && cf.programmeNextAction
                ? text(existing, cf.programmeNextAction)
                : "";
        const priorVerification =
            existing && cf.nextVerificationGate
                ? text(existing, cf.nextVerificationGate)
                : "";

        if (cf.reconciliationException && !priorException) {
            fields[cf.reconciliationException.id] = {
                name: structuredHoldState.exception
            };
        }
        if (cf.programmeNextAction && !priorNext) {
            fields[cf.programmeNextAction.id] =
                structuredHoldState.next;
        }
        if (cf.nextVerificationGate && !priorVerification) {
            fields[cf.nextVerificationGate.id] =
                structuredHoldState.gate;
        }
    }

    if (cf.programmeGate && programmeSemanticHoldReason) {
        const currentGate =
            existing
                ? selectName(existing, cf.programmeGate)
                : "";

        if (
            [
                "",
                "Unassessed",
                "Ready for Controlled Master Review"
            ].includes(currentGate)
        ) {
            fields[cf.programmeGate.id] = {
                name: "Evidence Captured – Scope Unresolved"
            };
        }
    }

    fields[cf.asset.id] =
        clean(c.asset);

    fields[cf.molecule.id] =
        clean(c.molecule);

    fields[cf.devCode.id] =
        clean(c.developmentCode);

    fields[cf.brand.id] =
        clean(c.brand);

    fields[cf.indication.id] =
        clean(c.indication);

    fields[cf.canonicalIndication.id] =
        clean(
            c.controlledIndicationCandidate
        );

    fields[cf.phase.id] =
        clean(c.phase);

    fields[cf.programStatus.id] =
        clean(c.programStatus);

    fields[cf.owner.id] =
        clean(c.sponsorOwner);

    fields[cf.partners.id] =
        Array.isArray(c.partners)
            ? c.partners
                .map(clean)
                .filter(Boolean)
                .join("; ")
            : "";

    fields[cf.sourceRecord.id] =
        clean(c.sourceRecordId);

    fields[cf.sourceUrl.id] =
        clean(c.sourceUrl);

    fields[cf.existingMatch.id] =
        (
            Array.isArray(
                c.existingPortfolioRecordIds
            )
                ? c.existingPortfolioRecordIds
                : []
        )
            .filter(
                x =>
                    /^rec[A-Za-z0-9]{14}$/.test(x)
            )
            .map(id => ({id}));

    fields[cf.matchMethod.id] =
        {
            name:
                mappedMatchMethod(
                    clean(c.matchMethod)
                )
        };

    fields[cf.confidence.id] =
        {
            name:
                clean(
                    c.matchConfidence ||
                    "Low"
                )
        };

    // Preserve a completed review's commercial decision as well as its
    // status, scope, gate, narrative and classification. A parser artefact
    // excluded from programme scope can never silently become Include.
    const priorReviewStatus = existing
        ? selectName(existing, cf.reviewStatus)
        : "";
    const priorDecision = existing
        ? selectName(existing, cf.decision)
        : "";
    const priorProgrammeGate = existing
        ? selectName(existing, cf.programmeGate)
        : "";
    const lockedReviewStatus = [
        "Reviewed - Include",
        "Reviewed - Exclude",
        "Resolved",
        "Superseded"
    ].includes(priorReviewStatus);
    const retainedDecision =
        (priorReviewStatus === "Reviewed - Exclude" ||
         priorProgrammeGate === "Out of Programme Scope")
            ? "Exclude"
            : (
                lockedReviewStatus &&
                ["Include", "Exclude", "Review"].includes(priorDecision)
                    ? priorDecision
                    : ""
            );

    fields[cf.decision.id] =
        {
            name: retainedDecision ||
                (semanticReview
                    ? "Review"
                    : clean(
                        c.commercialInclusionDecision ||
                        "Review"
                    ))
        };

    fields[cf.basis.id] =
        (
            Array.isArray(c.inclusionBasis)
                ? c.inclusionBasis
                : []
        )
            .map(x => clean(x))
            .filter(Boolean)
            .map(name => ({name}));

    fields[cf.reason.id] =
        semanticReview
            ? (
                "Semantic QA hold: " +
                semanticReview.reason +
                (
                    clean(c.reviewReason)
                        ? " | " + clean(c.reviewReason)
                        : ""
                )
            )
            : clean(c.reviewReason);

    fields[cf.evidence.id] =
        evidenceBits.join("\n");

    // Current machine comparator output has a dedicated overwrite-safe field.
    // Never replace a reviewer's previously appended notes during upsert.
    fields[cf.latestStagingEvidence.id] =
        evidenceBits.concat(
            clean(c.reviewReason)
                ? ["Latest comparator review reason: " + clean(c.reviewReason)]
                : []
        ).join("\n");

    fields[cf.batch.id] =
        batchRunId;

    fields[cf.lastChecked.id] =
        now;

    if (existing) {
        // Preserve both existing narratives, whether machine-originated or curated.
        // Fresh comparison evidence is still recorded in Latest Staging Evidence.
        delete fields[cf.reason.id];
        delete fields[cf.evidence.id];
        updates.push({
            id: existing.id,
            fields
        });
    } else {
        fields[cf.firstDetected.id] =
            now;

        creates.push({
            fields
        });
    }
}

const createdIds = [];

if (APPLY_STAGING_WRITES) {
    for (
        let i = 0;
        i < creates.length;
        i += 50
    ) {
        const ids =
            await candidateTable
                .createRecordsAsync(
                    creates.slice(i, i + 50)
                );

        if (Array.isArray(ids)) {
            createdIds.push(...ids);
        }
    }

    for (
        let i = 0;
        i < updates.length;
        i += 50
    ) {
        await candidateTable
            .updateRecordsAsync(
                updates.slice(i, i + 50)
            );
    }
}

/*
Bootstrap transition:
A newly validated canonical pipeline source may have a blank Pipeline Baseline
Status. Once and only once its adapter passes the same structural guardrails
and staging completes successfully, move that Source Watch record to
"Needs Review". This lets the existing Baseline Preflight -> Safety Gate
workflow pick it up without any company-specific scripting.
*/
let baselineStatusAfter =
    selectName(sw, swf.baselineStatus);

const sourceWatchSuccessFields = {};

if (["", "not established"].includes(baselineStatusBefore)) {
    sourceWatchSuccessFields[swf.baselineStatus.id] = {
        name: "Needs Review"
    };
    baselineStatusAfter = "Needs Review";
}

if (swf.lastChecked) {
    sourceWatchSuccessFields[swf.lastChecked.id] =
        now;
}

if (stagingStateNotesField) {
    const previous =
        String(
            sw.getCellValueAsString(
                stagingStateNotesField
            ) || ""
        ).trim();

    const retained =
        previous
            .split(/\n+/)
            .filter(
                line =>
                    !line.includes(
                        "PORTFOLIO_DISCOVERY_STAGING_V4.4.6 SUCCESS"
                    ) &&
                    !line.includes(
                        "PORTFOLIO_DISCOVERY_STAGING_V4.4.6 FAIL CLOSED"
                    )
            )
            .join("\n")
            .trim();

    const successDiagnostic =
        "PORTFOLIO_DISCOVERY_STAGING_V4.4.6 SUCCESS " +
        now +
        ": sourceRows=" +
        sourceRows.length +
        "; stagedCandidates=" +
        stageCandidates.length +
        "; semanticReviewRows=" +
        semanticReviewBySourceRecordId.size +
        "; created=" +
        creates.length +
        "; updated=" +
        updates.length;

    sourceWatchSuccessFields[
        stagingStateNotesField.id
    ] =
        (
            retained
                ? retained + "\n" + successDiagnostic
                : successDiagnostic
        ).slice(-10000);
}

if (
    APPLY_STAGING_WRITES &&
    Object.keys(sourceWatchSuccessFields).length > 0
) {
    await sourceWatchTable.updateRecordAsync(
        sw.id,
        sourceWatchSuccessFields
    );
}

const classificationCounts = {};

for (const c of comparison.candidates) {
    classificationCounts[c.classification] =
        (
            classificationCounts[
                c.classification
            ] || 0
        ) + 1;
}

output.set(
    "summary",
    JSON.stringify(
        {
            version: VERSION,
            company: companyName,
            sourceWatchRecordId,
            sourceName:
                text(sw, swf.sourceName),
            adapterProfile,
            retrievalMode,
            baselineStatusBefore:
                baselineStatusBefore || "BLANK",
            baselineStatusAfter,
            sourceRows:
                sourceRows.length,
            portfolioRows:
                portfolioRows.length,
            comparatorVersion:
                comparison.version,
            classificationCounts,
            stagedCandidates:
                stageCandidates.length,
            semanticReviewRows:
                semanticReviewBySourceRecordId.size,
            programmeSemanticHoldCount,
            programmeSemanticHoldReasons,
            plannedCreates:
                creates.length,
            plannedUpdates:
                updates.length,
            created:
                APPLY_STAGING_WRITES ? creates.length : 0,
            updated:
                APPLY_STAGING_WRITES ? updates.length : 0,
            stagingWritesEnabled:
                APPLY_STAGING_WRITES,
            createdRecordIds:
                createdIds,
            masterWrites: 0,
            status:
                "STAGING COMPLETE - Portfolio remains read-only"
        },
        null,
        2
    )
);

} catch (e) {
    const stagingError = String(
        (e && e.message) ||
        e ||
        "Unknown staging error"
    ).slice(0, 1200);

    /*
    Source-level observability for scale.

    Deterministic source/parser failures must not remain production-eligible.
    Structural and semantic contract failures quarantine only this source
    binding (not the reusable adapter globally) and reset its baseline to
    Not Established. Transient transport failures such as 429/5xx remain
    eligible for retry and are not auto-demoted.
    */
    const deterministicSourceFailure =
        /Source adapter failed staging guardrail|Semantic staging guardrail failed/i
            .test(stagingError);

    if (deterministicSourceFailure && APPLY_STAGING_WRITES) {
        try {
            const failureSw =
                await sourceWatchTable.selectRecordAsync(
                    sourceWatchRecordId
                );

            if (failureSw) {
                const bindingField =
                    sourceWatchTable.getField(
                        "Adapter Binding Status"
                    );
                const baselineField =
                    sourceWatchTable.getField(
                        "Pipeline Baseline Status"
                    );

                await sourceWatchTable.updateRecordAsync(
                    failureSw.id,
                    {
                        [bindingField.id]: {
                            name: "Needs Review"
                        },
                        [baselineField.id]: {
                            name: "Not Established"
                        }
                    }
                );
            }
        } catch (_) {
            // Quarantine bookkeeping failure must not change fail-closed behavior.
        }
    }

    if (stagingStateNotesField) {
        try {
            const failureSw =
                await sourceWatchTable.selectRecordAsync(
                    sourceWatchRecordId
                );

            if (failureSw) {
                const previous =
                    String(
                        failureSw.getCellValueAsString(
                            stagingStateNotesField
                        ) || ""
                    ).trim();

                const retained =
                    previous
                        .split(/\n+/)
                        .filter(
                            line =>
                                !line.includes(
                                    "PORTFOLIO_DISCOVERY_STAGING_V4.4.6 FAIL CLOSED"
                                )
                        )
                        .join("\n")
                        .trim();

                const diagnostic =
                    "PORTFOLIO_DISCOVERY_STAGING_V4.4.6 FAIL CLOSED " +
                    new Date().toISOString() +
                    ": " +
                    stagingError;

                const nextNotes =
                    (retained ? retained + "\n" : "") +
                    diagnostic;

                if (APPLY_STAGING_WRITES) {
                    await sourceWatchTable.updateRecordAsync(
                        failureSw.id,
                        {
                            [stagingStateNotesField.id]:
                                nextNotes.slice(-10000)
                        }
                    );
                }
            }
        } catch (_) {
            // Observability failure must never convert a safe source skip
            // into an automation-level failure.
        }
    }

    output.set(
        "summary",
        JSON.stringify(
            {
                version: VERSION,
                sourceWatchRecordId,
                error: stagingError,
                deterministicSourceFailure,
                sourceBindingQuarantined:
                    deterministicSourceFailure && APPLY_STAGING_WRITES,
                masterWrites: 0,
                status:
                    "FAIL CLOSED - SOURCE SKIPPED; AUTOMATION CONTINUES"
            },
            null,
            2
        )
    );
}


output.set(
    "canonicalPreview",
    JSON.stringify({
        version: VERSION,
        mode: "READ_ONLY_STABLE_KEY_IDENTITY_SHADOW",
        stagingWritesEnabled: APPLY_STAGING_WRITES,
        actualRecordWrites: APPLY_STAGING_WRITES ? "POSSIBLE" : 0,
        productionInstalled: false
    }, null, 2)
);
